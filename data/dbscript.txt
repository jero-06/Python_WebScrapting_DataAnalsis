# Database 생성 ( root password maria)
mysql -u root –p
    # database 목록 확인하기
    show databases;

    # mysql DB를 사용하겠다
    use mysql;

    # python_db 라는 DB를 만들겠다
    # 생성하다 뭐를 이름은
    create database python_db;

    # 생성한 후 목록 다시 확인
    show databases;

    # user는 python, password는 python 계정을 생성하겠다
    create user 'python'@'%' identified by 'python';

    # python 계정에게 python_db에 대한 모든 권한을 허용한다
    grant all privileges on python_db.* to 'python'@'%';
    
    # grant 명령 후 부여한 권한을 DB에 즉시 적용하겠다
    flush privileges;

    # 계정(user)이 생성됐는지 확인, user가 접속 가능한 host 확인    
    select user, host from user;

    # root계정에서 빠져나오겠다
    exit;
